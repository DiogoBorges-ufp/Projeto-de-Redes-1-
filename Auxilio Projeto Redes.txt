Configuração Switch A1

enable
configure terminal

! 1. Criar VLANs da infraestrutura
vlan 10
 name Guest
vlan 20
 name Student
vlan 30
 name Staff_Server
vlan 99
 name Management
exit

! 2. Configurar Spanning-Tree (Rapid-PVST) e proteções globais
spanning-tree mode rapid-pvst
spanning-tree portfast default
spanning-tree portfast bpduguard default

! 3. Agregação EtherChannel LACP para CD1 (Port-Channel 1) e CD2 (Port-Channel 2)
interface range GigabitEthernet0/0 - 1
 channel-group 1 mode active

interface range GigabitEthernet0/2 - 3
 channel-group 2 mode active
exit

interface Port-channel1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate

interface Port-channel2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
exit

! 4. Configurar Portas de Acesso dos Clientes (VLAN 10 para PC1 e PC4)
interface range GigabitEthernet1/0 - 1
 switchport mode access
 switchport access vlan 10
 switchport nonegotiate
 spanning-tree portfast
 spanning-tree bpduguard enable
exit

! 5. Contramedidas de Segurança (DHCP Snooping contra Yersinia)
ip dhcp snooping
ip dhcp snooping vlan 10,20,30,99

! Confiar apenas nos Uplinks para CD1/CD2
interface range Port-channel1 , Port-channel2
 ip dhcp snooping trust

! Limitar taxa de pacotes DHCP nas portas dos utilizadores
interface range GigabitEthernet1/0 - 1
 ip dhcp snooping limit rate 15
exit

Configuração do CD1

enable
configure terminal

! 1. Activar o Roteamento IP (L3)
ip routing

! 2. Criar a estrutura de VLANs
vlan 10
 name Guest
vlan 20
 name Student
vlan 30
 name Staff_Server
vlan 99
 name Management
exit

! 3. Prioridades Spanning-Tree (Root Primary nas VLANs 10 e 30)
spanning-tree mode rapid-pvst
spanning-tree vlan 10,30 priority 4096
spanning-tree vlan 20,99 priority 8192

! 4. Configurar Interfaces L3 Ponto-a-Ponto Roteadas
interface GigabitEthernet0/0
 description Link_CD1_EdgeRouter
 no switchport
 ip address 10.50.15.1 255.255.255.252
 no shutdown

interface GigabitEthernet0/2
 description Link_Inter_CD1_CD2
 no switchport
 ip address 10.50.15.9 255.255.255.252
 no shutdown

! 5. Interfaces SVIs com VRRP e DHCP Relay (ip helper-address)
interface Vlan10
 ip address 10.50.0.130 255.255.255.192
 ip helper-address 10.50.0.250
 vrrp 10 ip 10.50.0.129
 vrrp 10 priority 110
 vrrp 10 authentication text SecretKEY
 no shutdown

interface Vlan20
 ip address 10.50.0.2 255.255.255.128
 ip helper-address 10.50.0.250
 vrrp 20 ip 10.50.0.1
 vrrp 20 priority 100
 vrrp 20 authentication text SecretKEY
 no shutdown

interface Vlan30
 ip address 10.50.0.194 255.255.255.224
 ip helper-address 10.50.0.250
 vrrp 30 ip 10.50.0.193
 vrrp 30 priority 110
 vrrp 30 authentication text SecretKEY
 no shutdown

interface Vlan99
 ip address 10.50.0.226 255.255.255.240
 ip helper-address 10.50.0.250
 vrrp 99 ip 10.50.0.225
 vrrp 99 priority 100
 vrrp 99 authentication text SecretKEY
 no shutdown

! 6. Interface Loopback e Roteamento OSPFv2 (Área 0)
interface Loopback0
 ip address 1.1.1.1 255.255.255.255

router ospf 1
 router-id 1.1.1.1
 passive-interface Vlan10
 passive-interface Vlan20
 passive-interface Vlan30
 passive-interface Vlan99
 network 10.50.15.0 0.0.0.3 area 0
 network 10.50.15.8 0.0.0.3 area 0
 network 10.50.0.0 0.0.15.255 area 0
exit

! 7. Guardar na memória NVRAM
end

Configuração do Edge Router

enable
configure terminal

! 1. Configurar Interface Loopback (Router ID)
interface Loopback0
 ip address 3.3.3.3 255.255.255.255
 no shutdown

! 2. Configurar Interface L3 para o Switch CD1
interface GigabitEthernet0/0
 description Link_Ponto_a_Ponto_CD1
 ip address 10.50.15.2 255.255.255.252
 ip nat inside
 no shutdown

! 3. Configurar Interface L3 para o Switch CD2
interface GigabitEthernet0/1
 description Link_Ponto_a_Ponto_CD2
 ip address 10.50.15.6 255.255.255.252
 ip nat inside
 no shutdown

! 4. Configurar Interface WAN (Ligação à Nuvem Internet nat0)
interface GigabitEthernet0/2
 description Saida_WAN_Internet
 ip address dhcp
 ip nat outside
 no shutdown

! 5. Roteamento Dinâmico OSPFv2 (Área 0) & Injeção da Rota por Defeito
router ospf 1
 router-id 3.3.3.3
 network 10.50.15.0 0.0.0.3 area 0
 network 10.50.15.4 0.0.0.3 area 0
 default-information originate

! 6. Configuração de NAT/PAT (Tradução de Endereços)
! Permite saída à Internet para toda a gama interna 10.50.0.0/20
access-list 100 permit ip 10.50.0.0 0.0.15.255 any
ip nat inside source list 100 interface GigabitEthernet0/2 overload

! Static NAT: Expor a porta 80 do Servidor Web Debian (10.50.0.250) para a WAN
ip nat inside source static tcp 10.50.0.250 80 interface GigabitEthernet0/2 80

! 7. ACL de Segurança L3/L4: Restrição da VLAN Guest (VLAN 10)
! Bloqueia acesso à rede interna e permite apenas tráfego para a Internet
ip access-list extended GUEST_RESTRICTIONS
 deny ip 10.50.0.128 0.0.0.63 10.50.0.0 0.0.15.255
 permit ip 10.50.0.128 0.0.0.63 any
exit

! Aplicar a ACL de entrada nas interfaces ligadas à distribuição
interface GigabitEthernet0/0
 ip access-group GUEST_RESTRICTIONS in

interface GigabitEthernet0/1
 ip access-group GUEST_RESTRICTIONS in

! 8. Guardar na memória NVRAM
end
write memory (wr)
